        </View>

              <View style={[styles.dot, { borderColor: step.reached ? resident.accentText : resident.border, backgroundColor: step.reached && !step.current ? resident.accent : resident.card }]}>
                {step.current ? <View style={[styles.currentDot, { backgroundColor: resident.accentText }]} />
                  : step.reached ? <Ionicons name="checkmark" size={14} color={resident.onAccent} /> : null}
              </View>
              {index < summary.steps.length - 1 ? <View style={[styles.line, { backgroundColor: step.reached && !step.current ? resident.accentText : resident.border }]} /> : null}
            </View>
            <View style={styles.stepBody}>
              <ThemedText accessibilityLabel={step.current ? `${step.label}, current stage` : step.label}
                style={[styles.stepLabel, { color: step.reached ? resident.text : resident.muted, fontWeight: step.current ? '700' : step.reached ? '600' : '500' }]}>{step.label}</ThemedText>
              <ThemedText style={[styles.date, { color: resident.muted }]}>{step.date ? homeTimestamp(step.date) : step.detail}</ThemedText>
            </View>
          </View>)}
        </View>
        </ScrollView>
        <View style={styles.footer}>
          <PrimaryButton presentation="resident" label="View full request" disabled={!visible} onPress={onViewFull} />
        </View>
      </Animated.View>
    </View>
  </Modal>;

function resolveFixtureSource(args) {
  let file;
  if (args.path.startsWith('@/')) file = path.join(app, 'src', args.path.slice(2));
  else if (path.isAbsolute(args.path)) file = args.path;
  else if (args.path.startsWith('.')) file = path.resolve(path.dirname(args.importer), args.path);
  else {
    const owner = path.isAbsolute(args.importer) ? args.importer : path.join(app, 'package.json');
    file = (args.path === 'react' || args.path.startsWith('react/') || args.path.startsWith('react-dom'))
      ? require.resolve(args.path) : createRequire(owner).resolve(args.path);
  }
  const candidates = [file, ...['.ts', '.tsx', '.js', '.jsx', '.json', '/index.ts', '/index.tsx', '/index.js'].map(extension => file + extension)];
  const found = candidates.find(candidate => fs.existsSync(candidate) && fs.statSync(candidate).isFile());
  if (!found) throw new Error(`Fixture source not found: ${args.path}`);
  const resolved = fs.realpathSync(found);
  if (!resolved.toLowerCase().startsWith(path.resolve(app, '../..').toLowerCase() + path.sep)) throw new Error(`Fixture source outside workspace: ${resolved}`);
  return { path: resolved, namespace: 'workspace-source' };
}
